COMMAND_PATTERNS = {
    # Browser/Web
    "open youtube": "https://youtube.com",
    "open google": "https://google.com",
    "open gmail": "https://gmail.com",
    "open github": "https://github.com",
    "open netflix": "https://netflix.com",
    "open spotify": "https://open.spotify.com",
    "open twitter": "https://twitter.com",
    "open instagram": "https://instagram.com",
    "open whatsapp": "https://web.whatsapp.com",
    "open linkedin": "https://linkedin.com",
    "open chatgpt": "https://chatgpt.com",
    "search": "https://google.com/search?q=",
}

def detect_command(self, text: str) -> dict | None:
    text_lower = text.lower().strip()
    
    # Open website commands
    for pattern, url in COMMAND_PATTERNS.items():
        if pattern in text_lower:
            if pattern == "search":
                query = text_lower.split("search")[-1].strip()
                return {
                    "type": "open_url",
                    "url": f"https://google.com/search?q={query}",
                    "reply": f"Opening search for: {query}"
                }
            return {
                "type": "open_url", 
                "url": url,
                "reply": f"Opening {pattern.replace('open ','')} for you."
            }
    
    # YouTube search
    if "youtube" in text_lower and ("search" in text_lower or "play" in text_lower or "find" in text_lower):
        query = text_lower.replace("youtube","").replace("search","").replace("play","").replace("find","").strip()
        return {
            "type": "open_url",
            "url": f"https://youtube.com/results?search_query={query}",
            "reply": f"Searching YouTube for: {query}"
        }
    
    return None
