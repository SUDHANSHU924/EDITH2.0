class EDITHMasterOrchestrator:
    def __init__(self):
        self.groq = Groq(
            api_key=os.getenv("GROQ_API_KEY")
        )
        self.async_groq = AsyncGroq(
            api_key=os.getenv("GROQ_API_KEY")
        )
        self.conversation_history = []
        self.active_system = "core"
        self.task_log = []
        self.current_task = None
        
    def detect_and_route(self, user_input: str) -> dict:
        """Detect language and route to correct system"""
        try:
            response = self.groq.chat.completions.create(
                model="llama-3.3-70b-versatile",
                messages=[{
                    "role": "user",
                    "content": ROUTING_PROMPT.format(
                        request=user_input
                    )
                }],
                max_tokens=200,
                temperature=0.1
            )
            content = response.choices[0].message.content
            start = content.find('{')
            end = content.rfind('}') + 1
            if start >= 0:
                return json.loads(content[start:end])
        except:
            pass
        return {
            "system": "core",
            "reason": "default",
            "subtask": user_input,
            "priority": "medium"
        }
    
    async def think_and_respond(
        self, 
        user_input: str
    ) -> dict:
        """Main orchestrator: route + respond"""
        
        # Route to correct system
        routing = self.detect_and_route(user_input)
        self.active_system = routing.get("system", "core")
        
        # Log task
        task_entry = {
            "id": len(self.task_log) + 1,
            "input": user_input,
            "system": self.active_system,
            "status": "processing",
            "timestamp": __import__('datetime')
                .datetime.now().isoformat()
        }
        self.task_log.append(task_entry)
        
        # Add to history
        self.conversation_history.append({
            "role": "user",
            "content": user_input
        })
        
        # Get system-specific context
        system_context = self._get_system_context(
            self.active_system
        )
        
        # Build messages
        messages = [
            {
                "role": "system",
                "content": MASTER_PROMPT + 
                    f"\n\nACTIVE SYSTEM: {system_context}"
            }
        ] + self.conversation_history[-15:]
        
        # Generate response
        try:
            response = self.groq.chat.completions.create(
                model="llama-3.3-70b-versatile",
                messages=messages,
                max_tokens=500,
                temperature=0.7
            )
            reply = response.choices[0].message.content
            
            # Clean for voice
            reply_clean = self._clean_text(reply)
            
            # Update history
            self.conversation_history.append({
                "role": "assistant",
                "content": reply_clean
            })
            
            # Update task log
            task_entry["status"] = "complete"
            task_entry["response"] = reply_clean[:100]
            
            return {
                "reply": reply_clean,
                "system": self.active_system,
                "routing": routing,
                "task_id": task_entry["id"]
            }
            
        except Exception as e:
            task_entry["status"] = "error"
            return {
                "reply": f"Error in orchestrator: {str(e)}",
                "system": "core",
                "routing": routing,
                "task_id": task_entry["id"]
            }
    
    async def stream_response(self, user_input: str):
        """Stream tokens for real-time response"""
        routing = self.detect_and_route(user_input)
        self.active_system = routing.get("system", "core")
        
        self.conversation_history.append({
            "role": "user",
            "content": user_input
        })
        
        system_context = self._get_system_context(
            self.active_system
        )
        
        messages = [
            {
                "role": "system", 
                "content": MASTER_PROMPT + 
                    f"\n\nACTIVE SYSTEM: {system_context}"
            }
        ] + self.conversation_history[-15:]
        
        full_response = ""
        
        try:
            stream = self.groq.chat.completions.create(
                model="llama-3.3-70b-versatile",
                messages=messages,
                max_tokens=500,
                temperature=0.7,
                stream=True
            )
            
            for chunk in stream:
                delta = chunk.choices[0].delta.content
                if delta:
                    full_response += delta
                    yield {
                        "token": delta,
                        "system": self.active_system,
                        "done": False
                    }
            
            self.conversation_history.append({
                "role": "assistant",
                "content": self._clean_text(full_response)
            })
            
            yield {
                "token": "",
                "system": self.active_system,
                "full_response": self._clean_text(
                    full_response
                ),
                "done": True
            }
            
        except Exception as e:
            yield {
                "token": f"Error: {str(e)[:50]}",
                "system": "core",
                "done": True
            }
    
    def _get_system_context(self, system: str) -> str:
        contexts = {
            "core": "Conversational AI. Be warm and helpful.",
            "planning": "Planning mode. Show structured steps: [OBJECTIVE] [PLAN] [EXECUTE]",
            "code": "Code mode. Write complete runnable code only. Add types and error handling.",
            "files": "File creation mode. Create complete documents.",
            "search": "Research mode. Format: [QUERY] [FINDINGS] [CONFIDENCE]",
            "learning": "Learning mode. Track patterns and update knowledge.",
            "ml": "Data science mode. Provide working Python ML code.",
            "iot": "IoT mode. Generate Home Assistant YAML configs.",
            "vision": "Vision mode. Analyze images in detail.",
            "voice": "Voice mode. Short sentences only. No markdown.",
            "personal": "Personal mode. Be empathetic. Detect emotions.",
            "security": "Security mode. Flag threats immediately.",
            "daily": "Daily ops. Be quick and helpful.",
            "hacker": "RESTRICTED. Authorized security ops only.",
            "satellite": "CLASSIFIED. High-level clearance required."
        }
        return contexts.get(system, "General assistant mode.")
    
    def _clean_text(self, text: str) -> str:
        import re
        text = re.sub(r'\*\*?(.*?)\*\*?', r'\1', text)
        text = re.sub(r'#{1,6}\s', '', text)
        text = re.sub(r'`{3}.*?`{3}', '', text, flags=re.DOTALL)
        text = re.sub(r'\n{3,}', '\n\n', text)
        return text.strip()
    
    def get_status(self) -> dict:
        return {
            "active_system": self.active_system,
            "conversation_turns": len([
                m for m in self.conversation_history 
                if m["role"] == "user"
            ]),
            "tasks_completed": len([
                t for t in self.task_log 
                if t["status"] == "complete"
            ]),
            "task_log": self.task_log[-10:],
            "systems_online": 15,
            "systems_locked": 2
        }
    
    def clear_history(self):
        self.conversation_history = []
        self.task_log = []

async def voice_orchestrator(
    audio: UploadFile = File(...),
    voice_response: bool = True
):
    """Voice in → EDITH thinks → Voice out"""
    from agents.master_orchestrator import orchestrator
    from groq import Groq
    import os
    
    groq = Groq(api_key=os.getenv("GROQ_API_KEY"))
    
    # Save audio
    with tempfile.NamedTemporaryFile(
        delete=False, suffix=".webm"
    ) as tmp:
        tmp.write(await audio.read())
        tmp_path = tmp.name
    
    try:
        # STT — Multilingual Whisper
        with open(tmp_path, "rb") as f:
            transcription = groq.audio.transcriptions.create(
                file=("audio.webm", f.read()),
                model="whisper-large-v3",
                response_format="text"
                # No language specified = auto detect
            )
        
        transcript = transcription.strip()
        if not transcript:
            return {"error": "Could not hear you"}
        
        # Think and respond
        result = await orchestrator.think_and_respond(
            transcript
        )
        
        response = {
            "transcript": transcript,
            "reply": result["reply"],
            "system": result["system"],
            "routing": result["routing"]
        }
        
        # TTS
        if voice_response:
            audio_bytes = await _text_to_speech(
                result["reply"]
            )
            response["audio_base64"] = base64.b64encode(
                audio_bytes
            ).decode()
        
        return response
        
    finally:
        os.unlink(tmp_path)
