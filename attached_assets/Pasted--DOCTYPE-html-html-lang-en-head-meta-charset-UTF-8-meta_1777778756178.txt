    <script>
        // --- CUSTOM CROSSHAIR & MOUSE PARALLAX ---
        const crosshair = document.getElementById('crosshair');
        const hudWrapper = document.getElementById('hud-wrapper');
        const initBtn = document.getElementById('init-btn');

        document.addEventListener('mousemove', (e) => {
            // Move Crosshair
            crosshair.style.left = e.clientX + 'px';
            crosshair.style.top = e.clientY + 'px';

            // Calculate Parallax
            const xOffset = (e.clientX / window.innerWidth - 0.5) * 40; // Max 40px shift
            const yOffset = (e.clientY / window.innerHeight - 0.5) * 40;
            
            hudWrapper.style.transform = `translate(${-xOffset}px, ${-yOffset}px)`;
        });

        // Crosshair interaction styling
        document.addEventListener('mousedown', () => crosshair.classList.add('active'));
        document.addEventListener('mouseup', () => crosshair.classList.remove('active'));
        initBtn.addEventListener('mouseenter', () => crosshair.classList.add('active'));
        initBtn.addEventListener('mouseleave', () => crosshair.classList.remove('active'));

        // --- TERMINAL BOOT SEQUENCE ---
        const bootScreen = document.getElementById('boot-screen');
        const bootText = document.getElementById('boot-text');
        const bootLines = [
            "STARK INDUSTRIES // TACTICAL OS V3.0",
            "LOADING KERNEL...",
            "BYPASSING MAINFRAME PROTOCOLS... [SUCCESS]",
            "ACCESSING ORBITAL SATELLITE NETWORK...",
            "CONNECTING TO GLOBAL DEFENSE GRID...",
            "INITIALIZING E.D.I.T.H. PROTOCOLS...",
            "SYSTEM READY."
        ];

        let lineIndex = 0;
        function runBootSequence() {
            if (lineIndex < bootLines.length) {
                bootText.innerHTML += bootLines[lineIndex] + "<br>";
                lineIndex++;
                setTimeout(runBootSequence, Math.random() * 300 + 200);
            } else {
                setTimeout(() => {
                    bootScreen.classList.add('hidden');
                    hudWrapper.classList.remove('hidden');
                }, 800);
            }
        }
        // Start Boot Sequence
        setTimeout(runBootSequence, 500);

        // --- THREE.JS GLOBAL SATELLITE CORE ---
        const tContainer = document.getElementById('three-container');
        const scene = new THREE.Scene();
        const camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);
        camera.position.z = 5;

        const renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        tContainer.appendChild(renderer.domElement);

        // Create Wireframe Globe (fits perfectly inside the audio ring)
        const globeGeo = new THREE.SphereGeometry(1.4, 24, 24);
        const globeMat = new THREE.MeshBasicMaterial({ 
            color: 0x00f3ff, 
            wireframe: true, 
            transparent: true, 
            opacity: 0.15 
        });
        const globe = new THREE.Mesh(globeGeo, globeMat);
        scene.add(globe);

        // Add orbiting nodes (satellites)
        const nodesGroup = new THREE.Group();
        const nodeGeo = new THREE.BoxGeometry(0.02, 0.02, 0.02);
        const nodeMat = new THREE.MeshBasicMaterial({ color: 0xffffff });
        for(let i=0; i<40; i++) {
            const node = new THREE.Mesh(nodeGeo, nodeMat);
            const radius = 1.45 + Math.random() * 0.1;
            const theta = Math.random() * Math.PI * 2;
            const phi = Math.acos((Math.random() * 2) - 1);
            node.position.x = radius * Math.sin(phi) * Math.cos(theta);
            node.position.y = radius * Math.sin(phi) * Math.sin(theta);
            node.position.z = radius * Math.cos(phi);
            nodesGroup.add(node);
        }
        scene.add(nodesGroup);

        function animateThreeJS() {
            requestAnimationFrame(animateThreeJS);
            globe.rotation.y += 0.002;
            globe.rotation.x += 0.001;
            nodesGroup.rotation.y -= 0.003;
            renderer.render(scene, camera);
        }
        animateThreeJS();

        // Handle Resize
        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
            canvas.width = 600;
            canvas.height = 600;
        });

        // --- DATA SIMULATION ---
        function generateHex() {
            let result = '';
            const chars = '0123456789ABCDEF';
            for (let i = 0; i < 200; i++) {
                result += chars.charAt(Math.floor(Math.random() * chars.length));
                if (i % 4 === 3) result += ' ';
            }
            return result;
        }

        setInterval(() => {
            document.getElementById('hex-1').innerText = generateHex();
            document.getElementById('hex-2').innerText = generateHex();
            
            if(isAudioActive) {
                document.getElementById('gps-cords').innerHTML = 
                    `LAT: ${(Math.random() * 90).toFixed(4)} N<br>LNG: ${(Math.random() * 180).toFixed(4)} W`;
            }
        }, 80);

        // --- CIRCULAR AUDIO WAVEFORM ---
        const canvas = document.getElementById('audio-canvas');
        const ctx = canvas.getContext('2d');
        const centerX = canvas.width / 2;
        const centerY = canvas.height / 2;
        const baseRadius = 200;

        let audioContext, analyser, dataArray;
        let isAudioActive = false;

        initBtn.addEventListener('click', async () => {
            initBtn.classList.add('hidden');
            document.getElementById('sys-status').innerText = 'ESTABLISHING SECURE HANDSHAKE...';
            
            try {
                const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
                audioContext = new (window.AudioContext || window.webkitAudioContext)();
                analyser = audioContext.createAnalyser();
                const source = audioContext.createMediaStreamSource(stream);
                
                source.connect(analyser);
                analyser.fftSize = 256; 
                dataArray = new Uint8Array(analyser.frequencyBinCount);
                
                isAudioActive = true;
                
                document.getElementById('sys-status').innerText = 'VOCAL RECOGNITION ACTIVE';
                document.getElementById('uplink-status').innerText = 'STARK SECURE (LINKED)';
                document.getElementById('bio-scan').innerText = 'PETER PARKER (VERIFIED)';
                document.getElementById('threat-lvl').innerText = 'MINIMAL';
                document.getElementById('threat-lvl').style.color = 'var(--cyan)';

                // Brighten the 3D globe on activation
                globeMat.opacity = 0.4;
                globeMat.color.setHex(0xffffff);
                setTimeout(() => globeMat.color.setHex(0x00f3ff), 500);

            } catch (err) {
                console.error("Mic access denied", err);
                document.getElementById('sys-status').innerText = 'AUDIO SENSOR FAILED - SIMULATION ACTIVE';
                document.getElementById('sys-status').style.color = 'var(--alert)';
                isAudioActive = 'simulated';
            }
        });

        function drawWaveform() {
            requestAnimationFrame(drawWaveform);
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            let audioSum = 0;

            if (isAudioActive === true) {
                analyser.getByteFrequencyData(dataArray);
            } else if (isAudioActive === 'simulated') {
                if(!dataArray) dataArray = new Uint8Array(128);
                for(let i=0; i<dataArray.length; i++) {
                    dataArray[i] = Math.max(0, Math.sin(Date.now() * 0.008 + i * 0.2) * 120 + (Math.random() * 40));
                }
            }

            const bufferLength = dataArray ? dataArray.length : 128;
            const sliceAngle = (Math.PI * 2) / bufferLength;

            // Draw Dual Spikes (Inner and Outer)
            ctx.beginPath();
            ctx.strokeStyle = 'rgba(0, 243, 255, 0.9)';
            ctx.lineWidth = 2;
            ctx.shadowBlur = 10;
            ctx.shadowColor = '#00f3ff';

            for (let i = 0; i < bufferLength; i++) {
                const value = dataArray ? dataArray[i] : 0;
                audioSum += value;
                
                const spike = (value / 255) * 60; // Max spike length 60px
                const outerRadius = baseRadius + spike;
                const innerRadius = baseRadius - (spike * 0.5); // Mirror inwards slightly
                const angle = i * sliceAngle;

                // Draw outward line
                const ox = centerX + Math.cos(angle) * outerRadius;
                const oy = centerY + Math.sin(angle) * outerRadius;
                // Draw inward line
                const ix = centerX + Math.cos(angle) * innerRadius;
                const iy = centerY + Math.sin(angle) * innerRadius;

                ctx.moveTo(ix, iy);
                ctx.lineTo(ox, oy);
            }
            ctx.stroke();
            ctx.shadowBlur = 0; // Reset

            // Scale UI on loud noises
            if(dataArray) {
                const averageVolume = audioSum / bufferLength;
                const scale = 1 + (averageVolume / 255) * 0.15;
                document.getElementById('hud-core').style.transform = `scale(${scale})`;
                
                // Spin globe faster on loud audio
                globe.rotation.y += (averageVolume / 255) * 0.05;
            }
        }
        drawWaveform();

    </script>
